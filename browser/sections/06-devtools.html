06.1 端到端实战：带本地持久化与事件委托的待办任务板

下面我们将前面学习的 HTML 结构、border-box 盒模型、DOM 动态渲染、事件委托（closest）与 localStorage 本地存储串联起来，编写一个零外部依赖、保存为 todo-app.html 即可直接运行的完整小应用：

todo-app.html · 综合运用 DOM 片段渲染、事件委托与 localStorage 持久化的完整示例 
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <title>浏览器实战 · 轻量待办清单</title>
  <style>
    * { box-sizing: border-box; }
    body { font-family: system-ui, sans-serif; max-width: 520px; margin: 32px auto; padding: 0 16px; }
    .input-row { display: flex; gap: 8px; margin-bottom: 16px; }
    .input-row input { flex: 1; padding: 8px 12px; border: 2px solid #14226b; border-radius: 6px; }
    .todo-item { display: flex; justify-content: space-between; padding: 10px; border-bottom: 1px solid #ddd; }
    .todo-item.done span { text-decoration: line-through; color: #888; }
  </style>
</head>
<body>
  <h2>📋 待办任务清单（刷新页面不丢失）</h2>
  <form id="todo-form" class="input-row">
    <input id="todo-input" placeholder="输入新任务后回车..." required>
    <button type="submit">添加任务</button>
  </form>
  <ul id="todo-list" style="list-style:none; padding:0;"></ul>

  <script>
    const STORAGE_KEY = 'garden_todos_v1';
    const formEl = document.getElementById('todo-form');
    const inputEl = document.getElementById('todo-input');
    const listEl = document.getElementById('todo-list');

    // 1. 从 localStorage 读取初始任务数组
    let todos = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');

    // 2. 统一保存并重新渲染 DOM 列表（使用 textContent 防止 XSS 脚本注入）
    function saveAndRender() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
      const fragment = document.createDocumentFragment();
      for (const item of todos) {
        const li = document.createElement('li');
        li.className = 'todo-item' + (item.done ? ' done' : '');
        li.dataset.id = item.id;

        const titleSpan = document.createElement('span');
        titleSpan.textContent = item.title;
        titleSpan.className = 'toggle-btn';
        titleSpan.style.cursor = 'pointer';

        const delBtn = document.createElement('button');
        delBtn.textContent = '删除';
        delBtn.className = 'del-btn';

        li.append(titleSpan, delBtn);
        fragment.appendChild(li);
      }
      listEl.replaceChildren(fragment); // 一次性替换全部子节点，仅触发一次布局更新
    }

    // 3. 监听表单提交：用 preventDefault 阻止表单默认刷新页面的行为
    formEl.addEventListener('submit', (event) => {
      event.preventDefault();
      todos.push({ id: String(Date.now()), title: inputEl.value.trim(), done: false });
      inputEl.value = '';
      saveAndRender();
    });

    // 4. 事件委托：只在父元素 <ul> 上绑定 1 个点击事件处理所有子项的切换与删除
    listEl.addEventListener('click', (event) => {
      const li = event.target.closest('.todo-item');
      if (!li) return;
      if (event.target.closest('.del-btn')) {
        todos = todos.filter(t => t.id !== li.dataset.id);
      } else if (event.target.closest('.toggle-btn')) {
        const target = todos.find(t => t.id === li.dataset.id);
        if (target) target.done = !target.done;
      }
      saveAndRender();
    });

    saveAndRender();
  </script>
</body>
</html>

06.2 网页常见异常与卡顿的四步排查手册

当线上页面或本地开发出现“打开慢、点击卡顿、样式错乱”时，不需要凭感觉盲目修改代码。打开 Chrome DevTools 按照下表对应的观测面板即可快速定位瓶颈：

观察到的故障现象 打开哪个 DevTools 面板？ 关键检查指标与改进措施
① 首屏打开慢，长时间白屏 Network 面板
（勾选 Disable cache 刷新）		 看请求瀑布流（Waterfall）：若是首字节等待时间（Waiting for server response / TTFB）过长，查后端接口耗时；若是某个 .js 下载耗时长且阻塞了 HTML 解析，为其加上 defer 或做代码拆包。
② 点击按钮或输入文字时明显卡顿掉帧 Performance 面板
（点击录制 → 触发操作 → 停止）		 观察 Main 主线程轨道上是否有带红色三角标记的长任务（Long Task，超过 50ms）。点开底部调用栈找到耗时最长的 JS 函数，减少循环内重复读取 offsetWidth 或将重计算移入 Web Worker。
③ 发版后用户看到的还是老版本页面 Network 面板
（查看 Size 列与响应头）		 检查主文档 index.html 的 Size 列是否显示 (disk cache) 且未发请求。若是，需将服务端对 .html 文件的响应头改为 Cache-Control: no-cache。
④ 图片加载出来的瞬间把下方文字往下顶（页面跳动） Elements 面板
（检查 <img> 标签属性）		 图片下载完成前浏览器不知道它的高度，原本高度为 0，下载完突然撑开导致大面积重排。解决：在 <img> 标签上显式写明 width 和 height 属性（或设置 CSS aspect-ratio）提前预留空间。
nav_timing.js · 在控制台一键打印当前页面的网络导航各阶段耗时（毫秒） 
// 在任意 HTTP/HTTPS 网页的 Console 面板粘贴执行，即可查看本次加载各阶段精确耗时
const nav = performance.getEntriesByType('navigation')[0];
if (nav) {
  console.table({
    '① DNS 域名解析耗时 (ms)': Math.round(nav.domainLookupEnd - nav.domainLookupStart),
    '② TCP + TLS 建连耗时 (ms)': Math.round(nav.connectEnd - nav.connectStart),
    '③ 请求发送至收到首字节 TTFB (ms)': Math.round(nav.responseStart - nav.requestStart),
    '④ HTML 文档下载耗时 (ms)': Math.round(nav.responseEnd - nav.responseStart),
    '⑤ DOM 解析与 defer 脚本完成 (ms)': Math.round(nav.domContentLoadedEventEnd - nav.responseEnd)
  });
}
