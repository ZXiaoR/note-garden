07.1 连接数据库：MySQL 连接池与 Redis 读写最小实战

  • 业务需求：根据用户 ID 查询账户信息（高频读走 Redis 缓存减轻数据库压力），以及安全执行用户余额扣减（走 MySQL 事务，防止 SQL 注入与并发连接耗尽）。

mysql_pool.mjs · 连接池、参数化查询与安全事务模板 [Node.js 20+] 
// 安装依赖: npm install mysql2
import mysql from 'mysql2/promise';

// 1. 进程启动时创建全局单例连接池（切勿在每次请求里 createConnection）
export const pool = mysql.createPool({
  host: process.env.DB_HOST || '127.0.0.1',
  user: process.env.DB_USER || 'root',
  password: process.env.DB_PASSWORD || 'DevPass_2026',
  database: 'shop',
  connectionLimit: 10,     // 最大并发连接数
  supportBigNumbers: true, // 防止 BIGINT 雪花 ID 超过 JS 安全整数丢精度
});

// 2. 普通单条查询：用 ? 占位符传参防 SQL 注入，pool.execute 会自动借还连接
export async function getUserById(userId) {
  const [rows] = await pool.execute(
    'SELECT id, nickname, balance FROM users WHERE id = ?',
    [userId]
  );
  return rows[0] || null;
}

// 3. 多步事务：手动 getConnection，且必须在 finally 中归还连接
export async function deductBalance(userId, amount) {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    const [res] = await conn.execute(
      'UPDATE users SET balance = balance - ? WHERE id = ? AND balance >= ?',
      [amount, userId, amount]
    );
    if (res.affectedRows === 0) throw new Error('余额不足');
    await conn.commit();
  } catch (err) {
    await conn.rollback();
    throw err;
  } finally {
    conn.release(); // 无论成功或抛错都必须归还，否则 10 次报错后整个连接池永久卡死
  }
}

07.2 · 登录鉴权：scrypt 密码加盐哈希、JWT 鉴权钩子中间件与 Redis Session 选型

07.2 登录鉴权：密码安全存储与 JWT 中间件拦截

  • 业务需求：实现用户注册/登录（密码绝不明文入库），登录成功后签发身份令牌；访问 /api/me 等受保护接口时通过中间件统一校验身份，未登录或令牌过期返回 401 Unauthorized。
  • 方案选型（JWT vs Redis Session）：
    • 无状态 JWT（JSON Web Token，JSON 网络令牌）：服务端把 { sub: userId, role } 加数字签名发给客户端，后续每次请求解析验签即可知道是谁，不用查库，适合多服务拆分与短效接口调用（例如有效期 15 分钟 ~ 2 小时）；
    • 有状态 Redis Session：如果业务要求「修改密码/封号后立即强制踢下线」或「限制同一账号最多 2 台设备同时在线」，纯 JWT 在过期前无法单方面作废，此时应改用「服务端生成随机 SessionID 存 Redis + 浏览器设 HttpOnly Cookie」（或短效 JWT 搭配存储在 Redis 里的可吊销 Refresh Token）。
auth_jwt.mjs · 内置 crypto.scrypt 密码哈希 + jose 签发校验 JWT 中间件 [Node.js 20+] 
// 安装依赖: npm install fastify jose
import Fastify from 'fastify';
import { randomBytes, scrypt, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';
import { SignJWT, jwtVerify } from 'jose';

const scryptAsync = promisify(scrypt);
// 生产环境必须从 process.env.JWT_SECRET 读取至少 32 字节的高熵随机密钥
const SECRET = new TextEncoder().encode(process.env.JWT_SECRET || 'replace-with-32-chars-env-secret!');

// 1. 密码加盐哈希（存入数据库格式: "随机盐hex:哈希值hex"）与恒定时间比对
async function hashPassword(plain) {
  const salt = randomBytes(16).toString('hex');
  const buf = await scryptAsync(plain, salt, 64);
  return `${salt}:${buf.toString('hex')}`;
}
async function verifyPassword(plain, stored) {
  const [salt, hashHex] = stored.split(':');
  const derived = await scryptAsync(plain, salt, 64);
  return timingSafeEqual(derived, Buffer.from(hashHex, 'hex')); // 防计时侧信道攻击
}

// 2. 鉴权中间件（Fastify preHandler 钩子）：提取 Bearer Token，验签后挂载到 request.user
async function requireAuth(request, reply) {
  const header = request.headers.authorization || '';
  const token = header.startsWith('Bearer ') ? header.slice(7) : '';
  if (!token) return reply.code(401).send({ code: 401, message: '未提供登录令牌' });

  try {
    const { payload } = await jwtVerify(token, SECRET, { algorithms: ['HS256'] });
    request.user = { id: payload.sub, role: payload.role };
  } catch {
    return reply.code(401).send({ code: 401, message: '登录已过期或令牌无效' });
  }
}

// 3. 登录接口签发 Token + 受保护接口挂载 { preHandler: [requireAuth] }
const app = Fastify();
const demoHash = await hashPassword('Pass_123456');

app.post('/api/login', async (request, reply) => {
  const ok = await verifyPassword(request.body.password, demoHash);
  if (!ok) return reply.code(401).send({ code: 401, message: '账号或密码错误' });

  const accessToken = await new SignJWT({ role: 'member' })
    .setProtectedHeader({ alg: 'HS256' })
    .setSubject('1001')
    .setIssuedAt()
    .setExpirationTime('2h')
    .sign(SECRET);
  return { code: 0, accessToken };
});

app.get('/api/me', { preHandler: [requireAuth] }, async (request) => {
  return { code: 0, user: request.user };
});

await app.listen({ port: 3000 });

常见坑与背后原理：① JWT 的 Payload 只是 Base64URL 编码（防篡改、不防偷看），切勿往里面塞密码、手机号或身份证号；② 不要用已过时的 MD5 / SHA1 存密码，必须用慢哈希 scrypt / argon2——注意 crypto.scrypt 在底层跑在 libuv 的 4 个工作线程上（见 原理篇 02.3）；③ Redis Session 滑动续期与同账号多端踢人完整设计见 《Redis 笔记 · 04.1 登录态与多端踢人》。

07.3 · 文件上传与大文件下载：使用 pipeline 流式落盘与响应，杜绝内存暴涨

07.3 文件上传与大文件下载（用流，不要一次读进内存）

  • 业务需求：容器只有 512MB 内存，需要支持用户上传几百 MB 的安装包/视频并安全落盘（或直传对象存储），同时提供大文件流式下载接口。
  • 技术方案：上传用 @fastify/multipart 拿到可读流 data.file，通过 pipeline(data.file, createWriteStream(safePath)) 边收边写磁盘，随机生成 UUID 文件名替代原文件名；下载时先 stat 获取文件大小写入 Content-Length，再直接将 createReadStream 交给响应流发送。
file_stream_api.mjs · 流式上传（带大小截断清理与防目录穿越）与流式下载 [Node.js 20.11+] 
// 安装依赖: npm install fastify @fastify/multipart
import Fastify from 'fastify';
import multipart from '@fastify/multipart';
import { createReadStream, createWriteStream } from 'node:fs';
import { mkdir, stat, unlink } from 'node:fs/promises';
import path from 'node:path';
import crypto from 'node:crypto';
import { pipeline } from 'node:stream/promises';

const UPLOAD_DIR = path.join(import.meta.dirname, 'uploads');
await mkdir(UPLOAD_DIR, { recursive: true });

const app = Fastify();
// 限制单次请求最多 1 个文件、单文件最大 200MB
await app.register(multipart, { limits: { files: 1, fileSize: 200 * 1024 * 1024 } });

// 1. 流式上传接口: curl -F "file=@./big.zip" http://localhost:3000/api/upload
const ALLOWED_EXT = new Set(['.zip', '.pdf', '.mp4', '.png']);
app.post('/api/upload', async (request, reply) => {
  const part = await request.file();
  if (!part) return reply.code(400).send({ message: '未找到上传文件' });

  const ext = path.extname(part.filename).toLowerCase();
  if (!ALLOWED_EXT.has(ext)) {
    part.file.resume(); // 必写：排空丢弃未消费的请求流，否则连接会挂死
    return reply.code(400).send({ message: `不支持的文件类型: ${ext}` });
  }

  // 绝不直接用用户传来的文件名落盘（防 ../../ 路径穿越），改用随机 UUID
  const savedName = `${crypto.randomUUID()}${ext}`;
  const destPath = path.join(UPLOAD_DIR, savedName);
  await pipeline(part.file, createWriteStream(destPath));

  // 若上传体积超出 200MB 上限，流会被截断并标记 truncated = true，需删掉残缺文件
  if (part.file.truncated) {
    await unlink(destPath).catch(() => {});
    return reply.code(413).send({ message: '文件大小超过 200MB 限制' });
  }
  return { code: 0, fileId: savedName };
});

// 2. 大文件流式下载接口: GET /api/download/:fileId
app.get('/api/download/:fileId', async (request, reply) => {
  // path.basename 剥离一切目录层级，彻底阻断路径穿越读取任意系统文件
  const safeName = path.basename(request.params.fileId);
  const targetPath = path.join(UPLOAD_DIR, safeName);

  const info = await stat(targetPath).catch(() => null);
  if (!info || !info.isFile()) return reply.code(404).send({ message: '文件不存在' });

  reply.header('Content-Length', info.size);
  reply.header('Content-Type', 'application/octet-stream');
  reply.header('Content-Disposition', `attachment; filename="${encodeURIComponent(safeName)}"`);
  return reply.send(createReadStream(targetPath)); // Fastify 内部自动托管流背压与异常销毁
});

await app.listen({ port: 3000 });

常见坑与背后原理：① 切忌调用 await part.toBuffer() 把大文件整块塞进内存；② 生产多实例部署时不要把文件存在容器本地硬盘（容器重启就丢且其他实例读不到），可把上方 createWriteStream 换成阿里云 OSS / AWS S3 的流式上传，或由后端签发预签名直传 URL（Presigned URL）让浏览器直接传对象存储；③ 流背压如何自动控速详见 《核心原理篇 · 03.2 Stream 背压状态机》 ↗。

07.4 · SSE 流式输出：服务端边生成边推送（转发大模型回答）与客户端断开联动取消

07.4 SSE 流式输出：边生成边推，客户端断开立即取消上游请求

  • 业务需求：AI 对话接口调用上游大模型需要 15 秒才能生成完整回答，要求每生成几个字就立刻推给前端渲染；同时如果用户看两句就关掉了网页或点了“停止生成”，后端必须立刻中断向上游大模型的请求，停止继续计费。
  • 技术方案：相比双向的 WebSocket，「客户端发一次提问 → 服务端单向持续吐数据」最轻量的标准方案是 SSE（Server-Sent Events，服务器推送事件）——本质就是一个不急着调 res.end() 的普通 HTTP 响应，设置 Content-Type: text/event-stream 后按 data: JSON字符串\n\n（注意结尾两个换行符标志一条消息结束）逐块写出；并把 req.on('close') 绑定到上游 fetch 的 AbortController 上。
sse_llm_proxy.mjs · SSE 流式转发上游大模型 + 客户端断连自动 abort 上游 [Node.js 20+] 
import http from 'node:http';
import { setTimeout as sleep } from 'node:timers/promises';

// 模拟或替换成真实上游流式接口（真实环境中直接用下方注释里的 fetch(LLM_URL, { signal })）
async function* streamFromUpstream(prompt, signal) {
  /* 真实转发大模型写法：
  const upstream = await fetch('https://api.openai.com/v1/chat/completions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${process.env.LLM_KEY}` },
    body: JSON.stringify({ model: 'gpt-4o-mini', stream: true, messages: [{ role: 'user', content: prompt }] }),
    signal, // 核心：客户端一断连，这里立即中断上游 TCP 连接停止烧 Token
  });

  for await (const chunk of upstream.body) {
    yield decoder.decode(chunk, { stream: true }); // { stream: true } 防止跨块多字节中文乱码
  } */
  const tokens = ['你好，', '这是一段', '通过 SSE ', '实时推送的流式回答。'];
  for (const token of tokens) {
    // sleep 传入 signal：一旦客户端断开触发 abort，立即抛出 AbortError 结束生成器
    await sleep(400, undefined, { signal });
    yield token;
  }
}

const server = http.createServer(async (req, res) => {
  if (req.url !== '/api/chat/stream') return res.writeHead(404).end();

  // 1. 设置标准 SSE 响应头（X-Accel-Buffering: no 告诉反向代理 Nginx 不要攒包缓冲，立即透传）
  res.writeHead(200, {
    'Content-Type': 'text/event-stream; charset=utf-8',
    'Cache-Control': 'no-cache, no-transform',
    'Connection': 'keep-alive',
    'X-Accel-Buffering': 'no',
  });
  res.flushHeaders(); // 立即把响应头推出去建立流连接

  // 2. 监听客户端连接关闭：若响应尚未正常写完就断开了，立即 abort 上游请求
  const upstreamCtrl = new AbortController();
  req.on('close', () => {
    if (!res.writableEnded) {
      console.log('[SSE] 客户端提前断开，正在取消上游生成任务...');
      upstreamCtrl.abort();
    }
  });

  // 3. 逐块写出 SSE 帧（每帧以 \n\n 结尾），全部完成后发送结束标记
  try {
    for await (const text of streamFromUpstream('hi', upstreamCtrl.signal)) {
      res.write(`data: ${JSON.stringify({ delta: text })}\n\n`);
    }
    res.write('data: [DONE]\n\n');
  } catch (err) {
    if (err.name !== 'AbortError') console.error('[SSE 上游异常]:', err.message);
  } finally {
    res.end();
  }
});

server.listen(3000, () => console.log('测试命令: curl -N http://localhost:3000/api/chat/stream'));

常见坑与背后原理：① 本地直连实时逐字出，一上生产变成等 15 秒一次性蹦出来——99% 是漏了响应头 X-Accel-Buffering: no 被 Nginx 缓存了；② 长耗时生成期间若超过 15 秒无新字节产出，建议每 15 秒发一行注释心跳 res.write(': ping\n\n') 防网关空闲超时切断连接；③ 单线程高效维持数万 SSE 长连接的内核机制见 《核心原理篇 · 00.1 非阻塞 I/O 与 epoll 多路复用》 ↗（流式断连续传与幂等进阶见 《SSE · 幂等 · 上下文压缩》）。

07.5 · 调用第三方接口：超时中断、指数退避抖动重试与并发限制封装

07.5 调用第三方接口：超时、重试与并发限制一体化

  • 业务需求：调用外部物流轨迹或短信网关，网络偶发抖动或对方短暂报 503 时希望自动重试自愈，但不能无休止死等，也不能把刚恢复的下游冲垮。
  • 技术方案：封装通用的 fetchWithRetry——① 每次重试单独绑定超时时间（如单次 2500ms）；② 严格区分可重试错误（仅对网络断连、超时、429 限流与 502/503/504 网关错误重试，400/401/403/404 绝不重试）；③ 采用指数退避 + 随机抖动（Exponential Backoff + Jitter）拉开重试间隔；④ 批量调用时配合 03.3 的 mapLimit 控住并发上限。
resilient_fetch.mjs · 带单次超时、状态码过滤与指数退避抖动重试的请求客户端 [Node.js 20+] 
import { setTimeout as sleep } from 'node:timers/promises';

const RETRYABLE_STATUS = new Set([429, 502, 503, 504]);

export async function fetchWithRetry(url, options = {}, { retries = 2, timeoutMs = 2500 } = {}) {
  for (let attempt = 0; attempt <= retries; attempt++) {
    try {
      const res = await fetch(url, {
        ...options,
        signal: AbortSignal.timeout(timeoutMs), // 每次尝试独立的超时计时器
      });

      if (res.ok) return await res.json();

      // 若是 400 参数错或 401 鉴权错，重试 100 次结果也一样，直接抛错快速失败
      if (!RETRYABLE_STATUS.has(res.status) || attempt === retries) {
        throw new Error(`HTTP ${res.status}: ${url}`);
      }
    } catch (err) {
      // 达到最大重试次数，或不是网络/超时类异常时向上抛出
      const isNetOrTimeout = err.name === 'TimeoutError' || err.name === 'TypeError';
      if (attempt === retries || (!isNetOrTimeout && !err.message.startsWith('HTTP 5') && !err.message.startsWith('HTTP 429'))) {
        throw err;
      }
    }
    // 指数退避 + 随机抖动：第 1 次等 ~200ms，第 2 次等 ~400ms，打散并发洪峰
    const backoffMs = 200 * (2 ** attempt) + Math.floor(Math.random() * 100);
    await sleep(backoffMs);
  }
}

常见坑与背后原理：① 非幂等写接口（如扣款、发短信）绝不能裸重试——万一第一次请求对方其实已经成功扣款了，只是回包路上网络断了，你一重试就扣了两次钱；写接口必须先在请求头带上唯一业务单号 Idempotency-Key 且确认下游支持幂等去重才能开重试；② 高频调外部域名时注意 dns.lookup 争抢 4 个工作线程的问题，详见 《核心原理篇 · 02.3 libuv 线程池与 DNS 缓存优化》 ↗。

07.6 · 定时任务与后台任务：进程内定时器的边界，以及什么时候必须交给消息队列

07.6 定时任务与后台任务：什么时候该交给消息队列

后端常有两类不在当前 HTTP 响应里同步干完的活：周期性定时任务（如每晚凌晨统计账单）与异步后台任务（如用户注册后发欢迎邮件、导出百万行 Excel）。如果统统用 setInterval / setTimeout 或轻量库 node-cron 在 Web 进程内存里硬跑，一旦服务重启或多开几个容器实例就会出事故。先看选型分界线：

任务特征 进程内定时器（node-cron + Redis 锁） 专业队列（BullMQ / Redis Stream / Kafka）
进程重启或发版时 内存态立即丢失（正在等的 setTimeout 直接蒸发） 持久化在 Redis / 磁盘日志中，新进程启动后接着消费，零丢失
多实例（如 4 台容器）部署 默认 4 台机器到点会同时跑 4 遍，必须额外套一层 Redis 分布式互斥锁 消费者组机制天然保证同一条任务只分给 1 台空闲 Worker 处理
失败重试与洪峰削峰 需纯手写，流量突增时会把 Web 主服务内存挤爆 内置最大并发度、指数退避重试、延迟投递与死信队列（DLQ）告警
明确选型结论 仅用于丢一次无碍、下个周期能自愈的轻量巡检（如每 5 分钟清理本地临时目录、刷新内存字典） 凡涉及发券通知、扣款对账、延迟 30 分钟关单、视频转码/大报表导出，一律交给外部队列
cron_with_lock.mjs · 多实例安全的定时巡检（防重入 + Redis SET NX EX 抢锁）[Node.js 20+] 
// 安装依赖: npm install node-cron ioredis
import cron from 'node-cron';
import Redis from 'ioredis';

const redis = new Redis(process.env.REDIS_URL || 'redis://127.0.0.1:6379');
let localRunning = false;

// 每 5 分钟执行一次（Cron 表达式: 分 时 日 月 周）
cron.schedule('*/5 * * * *', async () => {
  // 1. 单机防重入：如果上一轮任务卡住还没跑完，直接跳过本轮
  if (localRunning) return;

  // 2. 多实例互斥：多台容器同时到点时，只有抢到 240 秒租约锁的那 1 台执行
  const locked = await redis.set('lock:cron:sync-report', process.pid, 'NX', 'EX', 240);
  if (locked !== 'OK') return;

  localRunning = true;
  try {
    console.log(`[PID ${process.pid}] 抢锁成功，开始执行周期汇总...`);
    // await runSummaryJob();
  } catch (err) {
    console.error('[定时任务异常]:', err);
  } finally {
    localRunning = false;
  }
});

延伸阅读：在纯 Node.js 技术栈中若已有 Redis，可直接用成熟的 BullMQ 做持久任务队列；若需要跨语言解耦、百万级削峰、本地消息表或 30 分钟未支付延迟关单，详见 《消息队列笔记 · 03 七大常见业务实战场景》 与 《Redis 笔记 · 04.8 分布式锁安全解锁》。

07.7 · 写一个命令行工具（CLI）：#!/usr/bin/env node、util.parseArgs 与标准退出码

07.7 写一个命令行工具（CLI）：扫描大文件的零依赖实战

  • 业务需求：写一个团队内部命令 scan-large --dir ./dist --min-kb 500，找出构建产物里所有体积超标的大文件，超标时返回非 0 退出码阻断 CI 发版。
  • 技术方案：① 脚本首行写 #!/usr/bin/env node（告诉操作系统用环境变量里的 node 执行它），并在 package.json 的 "bin" 字段声明命令名；② 直接使用内置 util.parseArgs（Node.js 20+ 稳定，免装 commander / yargs） 解析选项；③ 正常数据输出到 stdout，错误告警输出到 stderr，用 process.exitCode = 1 标记检查未通过。
bin/scan-large.mjs · 零依赖可直接复制运行的 CLI 工具 [Node.js 20.11+] 
#!/usr/bin/env node
// package.json 中配置: "bin": { "scan-large": "./bin/scan-large.mjs" }
// 本地测试直接运行: node ./bin/scan-large.mjs --dir . --min-kb 10
import { parseArgs } from 'node:util';
import { readdir, stat } from 'node:fs/promises';
import path from 'node:path';

// 1. 使用内置 util.parseArgs 解析命令行长短选项
const { values } = parseArgs({
  options: {
    dir: { type: 'string', short: 'd', default: '.' },
    'min-kb': { type: 'string', short: 'm', default: '500' },
    help: { type: 'boolean', short: 'h', default: false },
  },
});

if (values.help) {
  console.log('用法: scan-large [--dir <目录>] [--min-kb <阈值KB>]');
  process.exit(0);
}

// 2. 递归扫描目录（Node 20+ readdir 原生支持 recursive: true，无需装 glob）
const rootDir = path.resolve(process.cwd(), values.dir);
const minBytes = Number(values['min-kb']) * 1024;
const entries = await readdir(rootDir, { withFileTypes: true, recursive: true });

const oversized = [];
for (const entry of entries) {
  if (!entry.isFile() || entry.parentPath.includes('node_modules')) continue;
  const fullPath = path.join(entry.parentPath, entry.name);
  const { size } = await stat(fullPath);
  if (size >= minBytes) {
    oversized.push({ file: path.relative(rootDir, fullPath), kb: (size / 1024).toFixed(1) });
  }
}

// 3. 区分标准输出与错误退出码：超标时设 exitCode = 1 通知 CI 流水线拦截
if (oversized.length > 0) {
  console.error(`发现 ${oversized.length} 个超过 ${values['min-kb']}KB 的文件:`);
  console.table(oversized);
  process.exitCode = 1;
} else {
  console.log('检查通过：未发现超标大文件');
}
CLI 发布的两个实用细节：① 在 macOS / Linux 下执行 chmod +x ./bin/scan-large.mjs 赋予可执行权限后，运行 npm link 即可在本机任意目录直接敲 scan-large 测试；② 在项目根目录加一行 .gitattributes（*.mjs text eol=lf）强制使用 Unix LF 换行符，防止在 Windows 上保存出 \r\n 导致 Linux 终端报 env: 'node\r': No such file or directory。

本章即时场景题（先想答案，再点击展开）

1. 用户在网页上问了 AI 一个很长的问题，看开头不对立刻关掉了网页，但后台监控显示上游 OpenAI 接口还是跑满了 20 秒才结束，钱照扣了，漏写了哪一步？

漏了将下行客户端连接关闭事件与上游请求的 AbortController 绑定。客户端关掉网页只是断开了「浏览器 ⇄ 你的 Node 服务」这段连接，你的 Node 服务发给 OpenAI 的那条 fetch 请求默认完全不知情，会继续收完所有数据（然后往已断开的 res 里空写丢弃）。必须在 SSE 入口创建 const ctrl = new AbortController() 传给上游 fetch(url, { signal: ctrl.signal })，并在 req.on('close', () => { if (!res.writableEnded) ctrl.abort(); }) 中立即中断上游。

2. 产品要求做一个「后台封禁违规账号后，该用户正在使用的 App 必须在 1 秒内全部失效无法调任何接口」，只用无状态 JWT 能做到吗？怎么改最轻量？

纯无状态 JWT 做不到，因为 JWT 一旦签发出去，只要签名对且没到 exp 过期时间，服务端中间件只做本地验签就会放行。最轻量的折中方案是「无状态 JWT + Redis 封禁黑名单（或用户版本号）」：封号或改密时往 Redis 写入 ban:uid:1001 = 1（TTL 设为与 JWT 剩余最大有效期相同，如 2 小时）；鉴权中间件验签通过后顺手花 0.2ms 查一次 EXISTS ban:uid:1001，命中则立即返回 401 踢下线，既保留了 JWT 的跨服务便利，又补齐了即时吊销能力。

3. 在下单成功接口最后写了 setTimeout(() => generatePdfAndSendEmail(orderId), 1000) 立即返回前端“下单成功”，上线一周后收到多起客诉没收到发票邮件，为什么？

主要有两个原因：① 发版或容器重启丢任务——只要在那 1 秒等待或正在生成 PDF 的几秒钟里运维恰好执行了滚动发版重启进程，内存里的定时任务就永久消失了；② 下游邮件服务偶发超时无重试——一旦邮件网关抖动报错，任务失败后没有任何持久化记录可供补偿重跑。这类跨网络、不允许丢失的异步后置动作，应写入 BullMQ / 消息队列（关键业务配合数据库本地消息表）由后台 Worker 可靠消费并自动重试。
