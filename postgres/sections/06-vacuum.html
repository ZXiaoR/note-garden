🛡️ 06.1 生产环境大表在线安全变更规范

在生产环境中对数百万至上亿行的大表执行表结构变更（DDL）或新建索引时，若直接使用默认语法，可能会长时间持有高级别表锁并阻塞线上正常的读写请求。日常运维发布数据库变更脚本时，应遵循以下四项安全规范：

🐘 safe_online_ddl.sql · 生产大表无锁变更标准写法 
-- 规范 1：执行 ALTER TABLE 前先设置锁等待超时（拿不到锁立即退出，不堵队列）
SET lock_timeout = '2s';

-- 规范 2：添加带常量默认值的新列（PG 11+ 仅修改系统元数据，毫秒级完成）
ALTER TABLE orders ADD COLUMN channel text NOT NULL DEFAULT 'web';

-- 规范 3：在线创建索引必须加上 CONCURRENTLY（全程不阻塞表的增删改查）
-- 注意：CONCURRENTLY 不能放在 BEGIN ... COMMIT 事务块内部执行
CREATE INDEX CONCURRENTLY idx_orders_channel ON orders (channel);

-- 规范 4：分两步无锁添加 CHECK 或外键约束
-- 第一步：加 NOT VALID（瞬间完成，仅对新写入行生效，暂不扫描历史数据）
ALTER TABLE orders
  ADD CONSTRAINT chk_amount_pos CHECK (amount >= 0) NOT VALID;

-- 第二步：执行 VALIDATE CONSTRAINT（仅持弱锁在线校验历史行，不阻塞读写）
ALTER TABLE orders VALIDATE CONSTRAINT chk_amount_pos;
📖 为什么要格外注意 lock_timeout 与 CONCURRENTLY？
  • 锁等待队列的连锁阻塞效应：ALTER TABLE 需要在极短瞬间获取表级排他锁（ACCESS EXCLUSIVE）。如果此时表上恰好有一条执行了 30 秒的慢查询未结束，ALTER TABLE 会排在慢查询后面等待；而后续到达的所有普通 SELECT 查询又会排在 ALTER TABLE 后面等待，导致整张表在 30 秒内无法读取。加上 SET lock_timeout = '2s' 后，若 2 秒内拿不到锁则立即放弃并稍后重试，保障线上业务平稳。
  • CREATE INDEX CONCURRENTLY 的注意事项：普通 CREATE INDEX 在构建期间会禁止对该表执行写入操作；加上 CONCURRENTLY 后会分两次扫描表并在后台合入增量变更，不阻塞线上读写。若中途因唯一键冲突或超时中断，会留下一个标记为 INVALID 的无效索引，需先执行 DROP INDEX CONCURRENTLY 索引名; 清理后再重新创建。

🌐 06.2 物化视图、声明式分区表与 pgvector 实战

1. 普通视图、物化视图与分区表适用场景

特性名称 核心机制与日常用途
普通视图
VIEW		 仅保存一段查询 SQL 定义，本身不占存储空间；每次查询视图时实时展开执行，用于简化复杂关联或隐藏敏感列。
物化视图
MATERIALIZED VIEW		 将查询结果真正持久化存储为物理表，并可在其上建立索引，非常适合耗时数秒的统计报表加速。需要更新数据时执行 REFRESH MATERIALIZED VIEW CONCURRENTLY 视图名;（带 CONCURRENTLY 刷新期间不阻塞读取，要求物化视图上有唯一索引）。
声明式分区表
PARTITION BY		 将上亿行的流水大表按时间范围（RANGE）、离散值列表（LIST）或哈希（HASH）拆分为多个物理子表。查询带上时间条件时自动触发分区裁剪（Partition Pruning）只扫目标月份子表；清理过期历史数据时直接 DETACH / DROP 对应月份子表即可瞬间释放磁盘空间。
🐘 partitioning_and_vector.sql · 范围分区表与 AI 向量检索 
-- 1. 创建按月范围分区的日志主表与月份子表
CREATE TABLE event_logs (
  id         bigint      GENERATED ALWAYS AS IDENTITY,
  event_type text        NOT NULL,
  payload    jsonb       NOT NULL DEFAULT '{}',
  created_at timestamptz NOT NULL,
  PRIMARY KEY (id, created_at) -- 分区表唯一约束必须包含分区键
) PARTITION BY RANGE (created_at);

CREATE TABLE event_logs_2026_11 PARTITION OF event_logs
  FOR VALUES FROM ('2026-11-01') TO ('2026-12-01');

-- 2. 启用 pgvector 扩展，实现 RAG 知识库向量相似度检索
CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE doc_chunks (
  id        bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  content   text   NOT NULL,
  embedding vector(1536) -- 存储 1536 维 Embedding 浮点向量
);

-- 建立 HNSW 近似最近邻索引，使用 <=> 余弦距离检索最相关的前 5 条文档
CREATE INDEX ON doc_chunks USING hnsw (embedding vector_cosine_ops);
SELECT id, content FROM doc_chunks
ORDER BY embedding <=> '[0.012, -0.034, ...]' LIMIT 5;

-- 1. 一键查询当前正在发生锁等待的会话及阻塞源头 PID 与 SQL
SELECT
  blocked.pid AS blocked_pid,
  blocked.query AS blocked_sql,
  pg_blocking_pids(blocked.pid) AS blocker_pids,
  now() - blocked.query_start AS wait_duration
FROM pg_stat_activity blocked
WHERE cardinality(pg_blocking_pids(blocked.pid)) > 0;

-- 2. 安全终止异常慢查询 (cancel) 或断开卡死连接 (terminate)
SELECT pg_cancel_backend(目标PID);
SELECT pg_terminate_backend(目标PID);

-- 3. 查看各张表的总大小（含索引与 TOAST）
SELECT relname,
       pg_size_pretty(pg_total_relation_size(oid)) AS total_size
FROM pg_class WHERE relkind = 'r' ORDER BY pg_total_relation_size(oid) DESC LIMIT 10;

🟢 order_service.ts · 后端使用 pg 连接池执行参数化事务 
import { Pool } from 'pg';

// 1. 初始化全局连接池（整个进程复用同一个 Pool 实例）
const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 20,                      // 最大连接数
  idleTimeoutMillis: 30_000,    // 空闲连接回收时间
  connectionTimeoutMillis: 3_000 // 获取连接超时时间
});

// 2. 端到端下单事务：必须从 pool 借出同一个 client 才能共享事务
export async function createOrder(userId: number, itemId: number, amount: string) {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    // 使用 $1, $2 参数化占位符防止 SQL 注入，原子扣减库存
    const stockRes = await client.query(
      'UPDATE items SET stock = stock - 1 WHERE id = $1 AND stock > 0 RETURNING id',
      [itemId]
    );
    if (stockRes.rowCount === 0) throw new Error('库存不足');

    // 插入订单并通过 RETURNING 直接拿回生成的订单行
    const orderRes = await client.query(
      'INSERT INTO orders (user_id, amount) VALUES ($1, $2) RETURNING *',
      [userId, amount]
    );

    await client.query('COMMIT');
    return orderRes.rows[0];
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release(); // 务必在 finally 中将连接归还连接池
  }
}
